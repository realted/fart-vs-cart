cmake_minimum_required(VERSION 3.16)
project(FpgaRayTest LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "The serial backend currently uses the Windows COM API.")
endif()
add_executable(fpga_ray_test fpga_ray_test.cpp)
target_compile_features(fpga_ray_test PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(fpga_ray_test PRIVATE /W4)
else()
    target_compile_options(fpga_ray_test PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME fpga_reference COMMAND fpga_ray_test --self-test)
